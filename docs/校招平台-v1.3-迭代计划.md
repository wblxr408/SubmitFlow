### 2.3 模块 C：推荐算法精确化（P1）

#### 2.3.1 当前算法问题
1. **城市匹配过于粗粒度**：只区分"目标城市"和"非目标城市"
2. **紧迫性评分跳跃过大**：3天/7天/14天/30天分段不够平滑
3. **知名度权重线性**：大厂和中厂得分差异不够明显
4. **缺少公司规模因子**：同等知名度下，大厂应优先

#### 2.3.2 算法优化方案

**城市匹配精细化**
```typescript
function calcCityScore(jobCity: string | null, targetCities: string[]): number {
  if (!jobCity) return 40;
  if (targetCities.length === 0) return 50;

  // 精确匹配
  if (targetCities.some(c => c.trim() === jobCity.trim())) return 100;

  // 一线城市互通（北京/上海/深圳/广州）
  const tier1Cities = ['北京', '上海', '深圳', '广州'];
  if (tier1Cities.includes(jobCity) && targetCities.some(c => tier1Cities.includes(c))) {
    return 85;
  }

  // 新一线城市
  const newTier1Cities = ['杭州', '成都', '南京', '武汉', '西安', '苏州', '长沙', '天津'];
  if (newTier1Cities.includes(jobCity) && targetCities.some(c => newTier1Cities.includes(c))) {
    return 80;
  }

  // 同区域
  const regionMap = {
    '华东': ['南京', '苏州', '杭州', '宁波', '无锡', '合肥', '厦门', '福州', '济南', '青岛'],
    '华南': ['广州', '深圳', '东莞', '佛山', '珠海', '南宁', '海口'],
    '华北': ['北京', '天津', '石家庄', '太原', '呼和浩特'],
    '华中': ['武汉', '长沙', '郑州', '南昌', '合肥'],
    '西南': ['成都', '重庆', '昆明', '贵阳'],
    '西北': ['西安', '兰州', '乌鲁木齐', '银川', '西宁'],
    '东北': ['沈阳', '大连', '长春', '哈尔滨', '吉林'],
  };

  const jobRegion = Object.entries(regionMap).find(([, cities]) =>
    cities.includes(jobCity)
  )?.[0];

  if (jobRegion && targetCities.some(c => {
    const targetRegion = Object.entries(regionMap).find(([, cities]) => cities.includes(c))?.[0];
    return targetRegion === jobRegion;
  })) {
    return 70;
  }

  // 可远程加分
  if (job.is_remote) return 75;

  return 30;
}
```

**紧迫性评分平滑化**
```typescript
function calcDeadlineScore(deadline: string | null): number {
  if (!deadline) return 30;

  const days = Math.ceil(
    (new Date(deadline).getTime() - Date.now()) / (1000 * 60 * 60 * 24)
  );

  if (days < 0) return 0;       // 已截止
  if (days === 0) return 100;   // 今天截止
  if (days === 1) return 95;    // 明天截止
  if (days <= 3) return 90;
  if (days <= 7) return 80;
  if (days <= 14) return 65;
  if (days <= 21) return 50;
  if (days <= 30) return 40;
  if (days <= 60) return 30;
  return 20;
}
```

**知名度非线性化**
```typescript
function calcFameScore(fameScore: number, companySize: string): number {
  // 基础分非线性映射
  const base = Math.pow(fameScore / 100, 0.8) * 100;

  // 公司规模加成
  const sizeBonus = {
    '巨头': 5,
    '大型': 3,
    '中型': 0,
    '小型': -2,
  }[companySize] ?? 0;

  return Math.min(100, Math.max(0, Math.round(base + sizeBonus)));
}
```

```typescript
function calcFreshnessScore(firstSeenAt: string): number {
  const days = Math.ceil(
    (Date.now() - new Date(firstSeenAt).getTime()) / (1000 * 60 * 60 * 24)
  );

  if (days <= 1) return 100;
  if (days <= 3) return 90;
  if (days <= 7) return 75;
  if (days <= 14) return 60;
  if (days <= 30) return 45;
  return 30;
}
```

```typescript
interface ExtendedWeights {
  fame_weight: number;
  match_weight: number;
  city_weight: number;
  deadline_weight: number;
  conversion_weight: number;
  freshness_weight: number;  // 新增：新鲜度

function calcCompositeScoreV2(job, tagPrefs, targetCities, weights): number {
  const {
    fame_weight,
    match_weight,
    city_weight,
    deadline_weight,
    conversion_weight,
    freshness_weight = 0.1,
  } = weights;

  const scores = [
    { score: calcFameScore(job.company_fame_score, job.company_size), weight: fame_weight },
    { score: calcMatchScore(job.tags, tagPrefs), weight: match_weight },
    { score: calcCityScore(job.city, targetCities, job.is_remote), weight: city_weight },
    { score: calcDeadlineScore(job.deadline), weight: deadline_weight },
    { score: calcFreshnessScore(job.first_seen_at), weight: freshness_weight },
  ];

  if (job.conversion_rate !== null) {
    scores.push({ score: calcConversionScore(job.conversion_rate), weight: conversion_weight });
  }

  const validScores = scores.filter(s => s.weight > 0);
  const validWeightSum = validScores.reduce((sum, s) => sum + s.weight, 0);

  if (validWeightSum <= 0) return 50;

  const weightedSum = validScores.reduce((sum, s) => sum + s.score * s.weight, 0);
  return Math.round(weightedSum / validWeightSum);
}
```

```typescript
// 岗位列表查询优化：使用 CTE 预计算
const optimizedJobsQuery = `
  WITH company_fames AS (
    SELECT id, name, fame_score FROM companies
  ),
  job_tags_agg AS (
    SELECT job_id,
           COALESCE(json_agg(jsonb_build_object(
             'id', t.id, 'label', t.label, 'color_hex', t.color_hex
           )) FILTER (WHERE t.id IS NOT NULL), '[]') AS tags
    FROM job_tags jt
    LEFT JOIN tags t ON t.id = jt.tag_id
    GROUP BY job_id
  )
  SELECT j.*,
         cf.name AS company_name,
         cf.fame_score AS company_fame_score,
         jta.tags,
         EXISTS (
           SELECT 1 FROM job_entrypoints je
           WHERE je.job_id = j.id AND je.status = 'active'
             AND je.entry_type <> 'official'
         ) AS has_referral
  FROM jobs j
  JOIN company_fames cf ON cf.id = j.company_id
  LEFT JOIN job_tags_agg jta ON jta.job_id = j.id
  WHERE j.status = 'valid'
  -- 筛选条件
  ORDER BY j.last_seen_at DESC
  LIMIT $1 OFFSET $2
`;
```

#### 2.4.4 缓存策略（轻量级）
```typescript
// src/lib/cache.ts
// 简单内存缓存，替代 Redis 接口预留
interface CacheEntry<T> {
  data: T;
  expiresAt: number;
}

class SimpleCache {
  private store = new Map<string, CacheEntry<unknown>>();

  get<T>(key: string): T | null {
    const entry = this.store.get(key);
    if (!entry) return null;
    if (Date.now() > entry.expiresAt) {
      this.store.delete(key);
      return null;
    }
    return entry.data as T;
  }

  set<T>(key: string, data: T, ttlSeconds: number): void {
    this.store.set(key, {
      data,
      expiresAt: Date.now() + ttlSeconds * 1000,
    });
  }

  invalidate(pattern: string): void {
    for (const key of this.store.keys()) {
      if (key.includes(pattern)) {
        this.store.delete(key);
      }
    }
  }
}

export const cache = new SimpleCache();

// 使用示例
```

### 3.2 认证中间件

```typescript
// src/middleware/auth.ts
import { NextResponse } from 'next/server';
import type { NextRequest } from 'next/server';
import jwt from 'jsonwebtoken';

const JWT_SECRET = process.env.JWT_SECRET!;
const PUBLIC_PATHS = ['/login', '/register', '/forgot-password', '/api/auth'];

export function authMiddleware(request: NextRequest) {
  const { pathname } = request.nextUrl;

  // 公开路径放行
  if (PUBLIC_PATHS.some(p => pathname.startsWith(p))) {
    return NextResponse.next();
  }

  // API 路径需要认证
  if (pathname.startsWith('/api/')) {
    const token = request.cookies.get('auth_token')?.value;

    if (!token) {
      return NextResponse.json({ error: 'Unauthorized' }, { status: 401 });
    }

    try {
      const decoded = jwt.verify(token, JWT_SECRET) as { userId: number };
      // 将 userId 注入到请求 headers
      const headers = new Headers(request.headers);
      headers.set('x-user-id', String(decoded.userId));
      return NextResponse.next({ request: { headers } });
    } catch {
      return NextResponse.json({ error: 'Invalid token' }, { status: 401 });
    }
  }

  // 页面路径重定向
  if (!token && !pathname.startsWith('/login')) {
    return NextResponse.redirect(new URL('/login', request.url));
  }

  return NextResponse.next();
}
```
